2 ЗАВДАННЯ
Розробити Bash-скрипт для взаємодії батьківського та дочірнього процесів за допомогою сигналів (SIGTERM, SIGUSR1, SIGUSR2). Скрипт має підтримувати локалізацію, обробку параметрів командного рядка (-h, --help), ведення локального журналу у форматі RFC 2822 та запис подій у системний журнал syslog.

3. ОПИС ВИКОНАНОЇ РОБОТИ
У межах роботи було реалізовано скрипт з архітектурою «батько-нащадок». Батьківський процес виконується в інтерактивному режимі, дозволяючи користувачеві надсилати сигнали дочірньому процесу, який працює у фоні.
Основні технічні рішення:
	реалізована через команду kill та перехоплення сигналів командою trap. Дочірній процес надсилає сигнал-підтвердження (ACK) батькові після кожної операції.
	нащадок ігнорує сигнал SIGHUP, що дозволяє йому продовжувати роботу після закриття батьківського процесу.
	функція write_log записує події у файл з розширенням .sh (згідно з ТЗ), використовуючи роздільник «;».
	додано функцію-заглушку для команди logger, що забезпечує працездатність скрипта у середовищі Git Bash на ОС Windows.

4. ВИСНОВКИ
Під час виконання роботи було опановано методи керування процесами в Unix-подібних системах. Набуто навичок роботи з асинхронними сигналами, обробки аргументів командного рядка та налаштування багаторівневого журналювання подій.

Приклади програмного коду
1  #!/bin/bash
2  
3  if ! command -v logger &> /dev/null; then
4    logger() {
5      return 0
6    }
7  fi
8  
9  LOG_DIR="$(dirname "$0")/log"
10  LOG_FILE="$LOG_DIR/unx-pzpiz-22-1-karpov-dmytro-lab1.sh"
11  
12  if [[ "$LANG" == *"uk"* || "$LANG" == *"UA"* ]]; then
13    ERR_PARAM="Помилка: Некоректний параметр"
14    HELP_MSG="Використання: $0 [-h|--help]\n  -h, --help  Показати цю довідку"
15    MSG_START="Запуск батьківського процесу"
16    MSG_CHILD_START="Дочірній процес запущено з PID:"
17    MSG_PROMPT="Оберіть дію (1: SIGUSR1, 2: SIGUSR2, 3: SIGTERM(вбити), 4: Статус, q: Вихід): "
18  else
19    ERR_PARAM="Error: Invalid parameter"
20    HELP_MSG="Usage: $0 [-h|--help]\n  -h, --help  Show this help"
21    MSG_START="Parent process started"
22    MSG_CHILD_START="Child process started with PID:"
23    MSG_PROMPT="Choose action (1: SIGUSR1, 2: SIGUSR2, 3: SIGTERM(kill), 4: Status, q: Quit): "
24  fi
25  
26  case "$1" in
27    -h|--help)
28      echo -e "$HELP_MSG"
29      exit 0
30      ;;
31    "")
32      ;;
33    *)
34      echo "$ERR_PARAM: $1" >&2
35      logger -t "lab1-karpov" "Error: Invalid parameter $1"
36      exit 1
37      ;;
38  esac
39  
40  mkdir -p "$LOG_DIR"
41  
42  write_log() {
43    local sig_num=$1
44    local sig_name=$2
45    local desc=$3
46    local date_str=$(LC_ALL=C date -R)
47    local timestamp=$(date +%s)
48    local pid=$$
49    echo "${date_str}; ${timestamp}; ${pid}; ${sig_num}; ${sig_name}; ${desc}" >> "$LOG_FILE"
50  }
51  
52  child_logic() {
53    local parent_pid=$1
54    trap '' SIGHUP
55    trap 'write_log "10" "SIGUSR1" "Child received SIGUSR1. Sending ACK."; kill -SIGUSR1 '$parent_pid'' SIGUSR1
56    trap 'write_log "12" "SIGUSR2" "Child received SIGUSR2. Sending ACK."; kill -SIGUSR2 '$parent_pid'' SIGUSR2
57    trap 'write_log "15" "SIGTERM" "Child received SIGTERM. Terminating."; logger -t "lab1-karpov" "Child $$ terminated"; exit 0' SIGTERM
58    write_log "0" "NONE" "Child process started"
59    logger -t "lab1-karpov" "Child process created with PID $$"
60    while true; do
61      sleep 2
62    done
63  }
64  
65  write_log "0" "NONE" "$MSG_START"
66  logger -t "lab1-karpov" "Script started. Parent PID: $$"
67  child_logic $$ &
68  CHILD_PID=$!
69  
70  trap 'echo -e "\n[!] Батько отримав підтвердження (SIGUSR1) від дочірнього процесу!"; write_log "10" "SIGUSR1" "Parent received ACK"' SIGUSR1
71  trap 'echo -e "\n[!] Батько отримав підтвердження (SIGUSR2) від дочірнього процесу!"; write_log "12" "SIGUSR2" "Parent received ACK"' SIGUSR2
72  
73  echo "$MSG_CHILD_START $CHILD_PID"
74  
75  while true; do
76    echo -n "$MSG_PROMPT"
77    read action
78    if ! kill -0 $CHILD_PID 2>/dev/null && [[ "$action" != "q" ]]; then
79      echo "Дочірній процес ($CHILD_PID) вже завершив роботу."
80      continue
81    fi
82    case "$action" in
83      1)
84        kill -SIGUSR1 $CHILD_PID
85        sleep 1
86        ;;
87      2)
88        kill -SIGUSR2 $CHILD_PID
89        sleep 1
90        ;;
91      3)
92        kill -SIGTERM $CHILD_PID
93        sleep 1
94        ;;
95      4)
96        echo "Статус дочірнього процесу ($CHILD_PID): ПРАЦЮЄ"
97        ;;
98      q)
99        exit 0
100       ;;
101   esac
102 done
